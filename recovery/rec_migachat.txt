function parseInline(text: string, kp: string): React.ReactNode[] {
  return text.split(/(\*\*[^*]+\*\*)/g).map((p, i) => {
    if (/^\*\*[^*]+\*\*$/.test(p)) {
      return (
        <Text key={`${kp}-${i}`} style={styles.mdBold}>
          {p.slice(2, -2)}
        </Text>

    }

  });

// Lightweight Markdown for assistant bubbles: headings, bullets, numbered
// lists, bold, blank-line spacing. (The model replies in Markdown.)
function RichMessage({ text }: { text: string }) {
  const lines = text.replace(/\r/g, '').split('\n');
  return (
    <View>
      {lines.map((line, i) => {
        if (!line.trim()) return <View key={i} style={{ height: 8 }} />;
        const heading = line.match(/^#{1,6}\s+(.*)/);
        const bullet = line.match(/^\s*[-*•]\s+(.*)/);
        const numbered = line.match(/^\s*(\d+)[.)]\s+(.*)/);
        if (heading) {
          return <Text key={i} style={styles.mdHeading}>{parseInline(heading[1], `h${i}`)}</Text>;
        }
        if (bullet) {
          return (
            <View key={i} style={styles.mdBulletRow}>
              <Text style={styles.mdBulletDot}>•</Text>
              <Text style={styles.mdBulletText}>{parseInline(bullet[1], `b${i}`)}</Text>
            </View>
          );
        }
        if (numbered) {
          return (
            <View key={i} style={styles.mdBulletRow}>
              <Text style={styles.mdNum}>{numbered[1]}.</Text>
              <Text style={styles.mdBulletText}>{parseInline(numbered[2], `n${i}`)}</Text>
            </View>
          );
        }
        return <Text key={i} style={styles.textAssistant}>{parseInline(line, `p${i}`)}</Text>;
      })}
    </View>
  );
}

103	function RichMessage({ text }: { text: string }) {
104	  // Strip the "AI-wrapper" em-dash / double-hyphen tell → natural commas.
105	  // (En-dash ranges like "Art. 36–51" are left intact.)
106	  let cleaned = text
107	    .replace(/\r/g, '')
108	    .replace(/\s*—\s*/g, ', ')
109	    .replace(/\s+--\s+/g, ', ');
110	  // While streaming, an opening ** may not have its closing ** yet — hide the
111	  // dangling marker so the user never sees raw "**" before it turns bold.
112	  if (((cleaned.match(/\*\*/g) || []).length) % 2 === 1) {
113	    const i = cleaned.lastIndexOf('**');
114	    cleaned = cleaned.slice(0, i) + cleaned.slice(i + 2);
115	  }
116	  const lines = cleaned.split('\n');
117	  return (
118	    <View>
119	      {lines.map((line, i) => {
120	        if (!line.trim()) return <View key={i} style={{ height: 8 }} />;
121	        const heading = line.match(/^#{1,6}\s+(.*)/);
122	        const bullet = line.match(/^\s*[-*•]\s+(.*)/);
123	        const numbered = line.match(/^\s*(\d+)[.)]\s+(.*)/);
124	        if (heading) {
125	          return <Text key={i} style={styles.mdHeading}>{parseInline(heading[1], `h${i}`)}</Text>;
126	        }
127	        if (bullet) {
128	          return (
129	            <View key={i} style={styles.mdBulletRow}>
130	              <Text style={styles.mdBulletDot}>•</Text>
131	              <Text style={styles.mdBulletText}>{parseInline(bullet[1], `b${i}`)}</Text>
132	            </View>
133	          );
134	        }
135	        if (numbered) {
136	          return (
137	            <View key={i} style={styles.mdBulletRow}>
138	              <Text style={styles.mdNum}>{numbered[1]}.</Text>
139	              <Text style={styles.mdBulletText}>{parseInline(numbered[2], `n${i}`)}</Text>
140	            </View>
141	          );
142	        }
143	        return <Text key={i} style={styles.textAssistant}>{parseInline(line, `p${i}`)}</Text>;
